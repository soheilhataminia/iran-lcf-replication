series lnCO2pc     = log(co2_pc)

series S_raw       = sanction

scalar Sbar        = @mean(S_raw)
scalar sS          = @stdev(S_raw)
series SIz         = (S_raw - Sbar) / sS

series INT         = SIz * lnEnergypc
series INT_CO2     = SIz * lnCO2pc

series post1979    = @recode(@year>=1979, 1, 0)

' Differences for unit root testing

series d_SIz         = d(SIz)
series d_lnCO2pc     = d(lnCO2pc)

' ------------------------------------------------------------
' 3) ECPI CONSTRUCTION (PCA-BASED ROBUSTNESS PROXY)
' ------------------------------------------------------------

series z_lnEnergypc = (lnEnergypc - @mean(lnEnergypc)) / @stdev(lnEnergypc)
series z_lnCO2pc    = (lnCO2pc    - @mean(lnCO2pc))    / @stdev(lnCO2pc)

group g_pca z_lnEnergypc z_lnCO2pc
sym COVPCA   = @cov(g_pca)
matrix EVEC   = @eigenvectors(COVPCA)

scalar w1 = EVEC(1,1)
scalar w2 = EVEC(2,1)

series ECPI = w1*z_lnEnergypc + w2*z_lnCO2pc

scalar corr_ecpi_energy = @cor(ECPI, lnEnergypc)
if corr_ecpi_energy < 0 then
    ECPI = -ECPI
endif

series INT_ECPI = SIz * ECPI
series d_ECPI   = d(ECPI)

scalar b_siz    = eq_main_uecm.@coefs(6)

scalar lamy      = b_lagY

scalar lr_const  = -b_c / lamy
scalar lr_gdp    = -b_gdp / lamy
scalar lr_gfcf   = -b_gfcf / lamy
scalar lr_energy = -b_energy / lamy
scalar lr_siz    = -b_siz / lamy
scalar lr_int    = -b_int / lamy

' --- Delta method SE: constant
matrix(2,2) Vc
Vc(1,1)=V_main(2,2)
Vc(1,2)=V_main(2,1)
Vc(2,1)=V_main(1,2)
Vc(2,2)=V_main(1,1)

vector(2) gc
gc(1)=b_c/(lamy^2)
gc(2)=-1/lamy

scalar se_const = @sqrt(@inner(gc, Vc*gc))
scalar z_const  = lr_const/se_const
scalar p_const  = 2*(1-@cnorm(@abs(z_const)))

' --- GDP
matrix(2,2) Vgdp
Vgdp(1,1)=V_main(2,2)
Vgdp(1,2)=V_main(2,3)
Vgdp(2,1)=V_main(3,2)
Vgdp(2,2)=V_main(3,3)

vector(2) ggdp
ggdp(1)=b_gdp/(lamy^2)
ggdp(2)=-1/lamy

scalar se_gdp = @sqrt(@inner(ggdp, Vgdp*ggdp))
scalar z_gdp  = lr_gdp/se_gdp
scalar p_gdp  = 2*(1-@cnorm(@abs(z_gdp)))

' --- GFCF
